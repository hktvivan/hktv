const HAC_EXT_FILTER_IDS = ["fsql", "impex", "groovy"];

/** `null` = show all sections; otherwise exactly one of fsql | impex | groovy. */
function hacExtLoadFilterId() {
  const allowed = new Set(HAC_EXT_FILTER_IDS);
  try {
    const v = localStorage.getItem(hacExtFilterKey());
    if (v == null || v === "" || v === "all") return null;
    if (v.startsWith("[")) {
      const arr = JSON.parse(v);
      if (Array.isArray(arr)) {
        for (const x of arr) {
          if (allowed.has(x)) return x;
        }
        return null;
      }
    }
    if (allowed.has(v)) return v;
  } catch (_) {}
  return null;
}

function hacExtSaveFilterId(id) {
  try {
    if (id == null || !HAC_EXT_FILTER_IDS.includes(id)) {
      localStorage.setItem(hacExtFilterKey(), "");
    } else {
      localStorage.setItem(hacExtFilterKey(), id);
    }
  } catch (_) {}
}

/** Debounced save on input; uses _hacExtSaveT on the element. */
function hacExtAttachFieldPersistence(inputEl, sectionId, argIndex) {
  const saved = hacExtLoadField(sectionId, argIndex);
  if (saved != null) inputEl.value = saved;

  inputEl.addEventListener("input", () => {
    clearTimeout(inputEl._hacExtSaveT);
    inputEl._hacExtSaveT = setTimeout(() => {
      hacExtSaveField(sectionId, argIndex, inputEl.value);
    }, 300);
  });
}

/** Title-style labels for placeholders: camelCase → words, then capitalize each token. */
function hacExtDisplayLabel(raw) {
  if (raw == null || raw === "") return "";
  const spaced = String(raw).replace(/([a-z])([A-Z])/g, "$1 $2");
  return spaced
    .split(/\s+/)
    .filter(Boolean)
    .map((w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase())
    .join(" ");
}

/** Placeholder like "User Pk (9064626978820)" when the callback uses a default for empty input. */
function hacExtPlaceholderWithDefault(rawField, defaultValue) {
  const base = hacExtDisplayLabel(rawField);
  if (defaultValue == null || defaultValue === "") return base;
  return `${base} (${defaultValue})`;
}

function hacExtSyncSectionSeparators() {
  let firstVisible = true;
  for (const sectionEl of EXTENSION_SECTIONS) {
    if (sectionEl.style.display === "none") continue;
    if (firstVisible) {
      sectionEl.style.borderTop = "";
      sectionEl.style.paddingTop = "";
      sectionEl.style.marginTop = "";
      firstVisible = false;
    } else {
      sectionEl.style.borderTop = `1px solid ${HAC_EXT_SEPARATOR}`;
      sectionEl.style.paddingTop = "10px";
      sectionEl.style.marginTop = "2px";
    }
  }
}

function hacExtApplySectionFilter(activeId) {
  const showAll = activeId == null;
  for (const sectionEl of EXTENSION_SECTIONS) {
    const tag = sectionEl.dataset.hacFilter;
    const show = showAll || tag === activeId;
    // Empty string clears inline display → computed `block`, so `gap` on sections stops working.
    if (show) {
      sectionEl.style.display = "flex";
      sectionEl.style.flexDirection = "column";
      sectionEl.style.alignItems = "stretch";
      sectionEl.style.width = "100%";
      sectionEl.style.boxSizing = "border-box";
      sectionEl.style.gap = HAC_EXT_STACK_GAP;
    } else {
      sectionEl.style.display = "none";
    }
  }
  hacExtSyncSectionSeparators();
}

async function searchSystemVariable() {
  let variableType = (document.querySelector("#searchSystemVariableArg0")?.value ?? "").trim();
  let variableCode = (document.querySelector("#searchSystemVariableArg1")?.value ?? "").trim();

  const lines = [
    "SELECT /*hktv-slave-db*/ {VariableType}, {VariableCode}, {VariableContent}",
    "FROM {SystemVariable}",
  ];

  const parts = [];
  if (variableType !== "") parts.push(`{VariableType} = "${variableType}"`);
  if (variableCode !== "") parts.push(`{VariableCode} = "${variableCode}"`);

  if (parts.length > 0) {
    lines.push(`WHERE ${parts[0]}`);
    for (let i = 1; i < parts.length; i++) {
      lines.push(`AND ${parts[i]}`);
    }
  }

  await hacExtCopyWithToast(lines.join("\n"));
}

function convertNameToId(name) {
  let words = name.toLowerCase().split(" ");
  let camelCase = words[0];

  for (let i = 1; i < words.length; i++) {
    let capitalizedWord = words[i].charAt(0).toUpperCase() + words[i].slice(1);
    camelCase += capitalizedWord;
  }

  return camelCase;
}
function createExtensionSection(name, args, callback, opts = {}) {
  let isShownBeforeLogin = false;
  let filter = "fsql";
  let fieldDefaults;
  if (typeof opts === "boolean") {
    isShownBeforeLogin = opts;
  } else if (opts && typeof opts === "object") {
    isShownBeforeLogin = Boolean(opts.isShownBeforeLogin);
    if (opts.filter) filter = opts.filter;
    fieldDefaults = opts.fieldDefaults;
  }

  const isInLoginPage = window.location.href.includes("login.jsp");

  if (isInLoginPage !== isShownBeforeLogin) return;

  const id = convertNameToId(name);

  const section = document.createElement("div");
  section.id = id;
  section.dataset.hacFilter = filter;
  hacExtApplyStyles(section, HAC_EXT_STYLES.section);

  const row = document.createElement("div");
  hacExtApplyStyles(row, HAC_EXT_STYLES.toolbarRow);

  const button = document.createElement("button");
  button.innerHTML = name;
  button.addEventListener("click", callback);
  button.type = "button";
  hacExtApplyStyles(button, HAC_EXT_STYLES.primaryButton);

  const resetBtn = document.createElement("button");
  resetBtn.type = "button";
  resetBtn.textContent = "Reset";
  resetBtn.setAttribute("data-hac-section-reset", "true");
  resetBtn.setAttribute("aria-label", `Reset ${name} fields`);
  hacExtApplyStyles(resetBtn, HAC_EXT_STYLES.resetButton);
  resetBtn.style.display = "none";

  resetBtn.addEventListener("click", () => {
    for (const [i] of args.entries()) {
      const el = document.querySelector(`#${id}Arg${i}`);
      if (el) el.value = "";
    }
    hacExtClearSectionFields(id, args.length);
    hacExtSyncSectionResetVisibility(section, resetBtn);
  });

  row.append(button, resetBtn);
  section.append(row);

  for (const [i, v] of args.entries()) {
    const inputBox = document.createElement("input");
    inputBox.id = id + `Arg${i}`;
    inputBox.type = "text";
    const defVal = fieldDefaults && fieldDefaults[i];
    inputBox.placeholder =
      defVal != null && defVal !== "" ? hacExtPlaceholderWithDefault(v, defVal) : hacExtDisplayLabel(v);
    hacExtApplyStyles(inputBox, HAC_EXT_STYLES.textInput);

    hacExtAttachFieldPersistence(inputBox, id, i);
    inputBox.addEventListener("input", () => {
      hacExtSyncSectionResetVisibility(section, resetBtn);
    });
    section.append(inputBox);
  }

  hacExtSyncSectionResetVisibility(section, resetBtn);

  EXTENSION_SECTIONS.push(section);

  return section;
}
